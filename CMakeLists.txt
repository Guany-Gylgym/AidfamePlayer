cmake_minimum_required(VERSION 3.24)
project(AidfamePlayer VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Svg SvgWidgets Test)
qt_standard_project_setup()

set(MPV_ROOT "${CMAKE_SOURCE_DIR}/vendor/mpv" CACHE PATH "Pinned libmpv SDK")
if(NOT EXISTS "${MPV_ROOT}/libmpv-2.dll")
    message(FATAL_ERROR "Run scripts/setup-mpv.ps1 first")
endif()
add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/mpv.lib"
    COMMAND "${CMAKE_AR}" /nologo /machine:x64 "/def:${CMAKE_SOURCE_DIR}/cmake/mpv.def" "/out:${CMAKE_BINARY_DIR}/mpv.lib"
    DEPENDS cmake/mpv.def VERBATIM)
add_custom_target(mpv_import DEPENDS "${CMAKE_BINARY_DIR}/mpv.lib")
add_library(player_core STATIC src/Player/MpvPlayer.cpp src/Player/MpvPlayer.h)
add_dependencies(player_core mpv_import)
target_include_directories(player_core PUBLIC src "${MPV_ROOT}/include")
target_link_libraries(player_core PUBLIC Qt6::Core "${CMAKE_BINARY_DIR}/mpv.lib")
if(MSVC)
    target_compile_options(player_core PUBLIC /W4 /permissive- /utf-8)
endif()

add_library(player_ui STATIC src/UI/MainWindow.cpp src/UI/MainWindow.h)
target_sources(player_ui PRIVATE src/UI/SettingsDialog.cpp src/UI/SettingsDialog.h)
target_sources(player_core PRIVATE src/Recorder/Recorder.cpp src/Recorder/Recorder.h)
target_sources(player_core PRIVATE src/Settings/LocalStore.cpp src/Settings/LocalStore.h)
target_sources(player_core PRIVATE src/Settings/CacheStore.cpp src/Settings/CacheStore.h)
target_sources(player_core PRIVATE src/Settings/FileAssociations.cpp src/Settings/FileAssociations.h)
target_link_libraries(player_core PUBLIC advapi32 shell32 runtimeobject ole32)
target_include_directories(player_ui PUBLIC src)
target_link_libraries(player_ui PUBLIC Qt6::Widgets Qt6::SvgWidgets player_core)
if(MSVC)
    target_compile_options(player_ui PUBLIC /W4 /permissive- /utf-8)
endif()

qt_add_executable(AidfamePlayer WIN32 src/App/main.cpp assets/resources.qrc assets/AidfamePlayer.rc)
target_link_libraries(AidfamePlayer PRIVATE player_ui)
add_custom_command(TARGET AidfamePlayer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${MPV_ROOT}/libmpv-2.dll" "$<TARGET_FILE_DIR:AidfamePlayer>" VERBATIM)
set(FFMPEG_EXE "${CMAKE_SOURCE_DIR}/.tools/python/Lib/site-packages/imageio_ffmpeg/binaries/ffmpeg-win-x86_64-v7.1.exe")
add_custom_command(TARGET AidfamePlayer POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${FFMPEG_EXE}" "$<TARGET_FILE_DIR:AidfamePlayer>/ffmpeg.exe" VERBATIM)

include(CTest)
if(BUILD_TESTING)
    qt_add_executable(ui_tests Tests/UiTests.cpp assets/resources.qrc)
    target_link_libraries(ui_tests PRIVATE player_ui Qt6::Test)
    target_compile_definitions(ui_tests PRIVATE MEDIA_DIR="${CMAKE_SOURCE_DIR}/Tests/Media")
    add_test(NAME ui_tests COMMAND ui_tests -o "${CMAKE_BINARY_DIR}/ui-test-results.txt,txt")
    set_tests_properties(ui_tests PROPERTIES TIMEOUT 60)
    qt_add_executable(playback_tests Tests/PlaybackTests.cpp)
    target_link_libraries(playback_tests PRIVATE player_core Qt6::Widgets Qt6::Test)
    set_target_properties(ui_tests playback_tests PROPERTIES WIN32_EXECUTABLE FALSE)
    target_compile_definitions(playback_tests PRIVATE MEDIA_DIR="${CMAKE_SOURCE_DIR}/Tests/Media")
    add_dependencies(ui_tests AidfamePlayer)
    add_dependencies(playback_tests AidfamePlayer)
    add_test(NAME playback_tests COMMAND playback_tests -o "${CMAKE_BINARY_DIR}/playback-test-results.txt,txt")
    set_tests_properties(playback_tests PROPERTIES TIMEOUT 120)
    qt_add_executable(recorder_tests Tests/RecorderTests.cpp)
    target_link_libraries(recorder_tests PRIVATE player_core Qt6::Widgets Qt6::Test)
    set_target_properties(recorder_tests PROPERTIES WIN32_EXECUTABLE FALSE)
    target_compile_definitions(recorder_tests PRIVATE MEDIA_DIR="${CMAKE_SOURCE_DIR}/Tests/Media"
        ENCODER_PATH="${CMAKE_SOURCE_DIR}/.tools/python/Lib/site-packages/imageio_ffmpeg/binaries/ffmpeg-win-x86_64-v7.1.exe")
    add_test(NAME recorder_tests COMMAND recorder_tests -o "${CMAKE_BINARY_DIR}/recorder-test-results.txt,txt")
    set_tests_properties(recorder_tests PROPERTIES TIMEOUT 300)
    qt_add_executable(settings_tests Tests/SettingsTests.cpp)
    target_link_libraries(settings_tests PRIVATE player_core Qt6::Test)
    set_target_properties(settings_tests PROPERTIES WIN32_EXECUTABLE FALSE)
    add_test(NAME settings_tests COMMAND settings_tests -o "${CMAKE_BINARY_DIR}/settings-test-results.txt,txt")
    qt_add_executable(cache_tests Tests/CacheTests.cpp)
    target_link_libraries(cache_tests PRIVATE player_core Qt6::Test)
    set_target_properties(cache_tests PROPERTIES WIN32_EXECUTABLE FALSE)
    add_test(NAME cache_tests COMMAND cache_tests -o "${CMAKE_BINARY_DIR}/cache-test-results.txt,txt")
    qt_add_executable(association_tests Tests/AssociationTests.cpp)
    target_link_libraries(association_tests PRIVATE player_core Qt6::Test)
    set_target_properties(association_tests PROPERTIES WIN32_EXECUTABLE FALSE)
    add_test(NAME association_tests COMMAND association_tests -o "${CMAKE_BINARY_DIR}/association-test-results.txt,txt")
    qt_add_executable(soak_runner Tests/SoakRunner.cpp)
    target_link_libraries(soak_runner PRIVATE player_core Qt6::Widgets psapi)
    set_target_properties(soak_runner PROPERTIES WIN32_EXECUTABLE FALSE EXCLUDE_FROM_ALL TRUE)
    qt_add_executable(media_tests Tests/MediaTests.cpp)
    target_link_libraries(media_tests PRIVATE player_core Qt6::Widgets Qt6::Test)
    set_target_properties(media_tests PROPERTIES WIN32_EXECUTABLE FALSE EXCLUDE_FROM_ALL TRUE)
endif()

install(TARGETS AidfamePlayer RUNTIME DESTINATION .)
install(FILES "${MPV_ROOT}/libmpv-2.dll" DESTINATION .)
install(PROGRAMS "${FFMPEG_EXE}" DESTINATION . RENAME ffmpeg.exe)
